// 示例思路
export default (initialState: API.UserInfo) => {
  const { routes = [] } = initialState || {};

  // 辅助函数：检查是否有路由权限
  const hasRoute = (routePath: string) => {
    return routes.includes(routePath);
  };

  // 辅助函数：检查是否有路由前缀权限（用于判断父路由）
  const hasRoutePrefix = (prefix: string) => {
    return routes.some((route) => route.startsWith(prefix));
  };

  return {
    // 页面级权限
    canAccessHome: hasRoute('/home'),
    canAccessSystem: hasRoute('/system'),
    canAccessOperation: hasRoute('/operation'),

    // System 页面子菜单权限
    canAccessUserProfile: hasRoute('/system/user-profile'),
    canAccessUserManagement: hasRoute('/system/user-management'),
    canAccessRoleManagement: hasRoute('/system/role-management'),
    canAccessResourceManagement: hasRoute('/system/resource-management'),

    // 操作级权限（如果后端也返回操作权限，可以扩展）
    // 或者约定：如果有 '/system/user-management' 权限，就默认有该模块的所有操作权限
    canAddUser: hasRoute('/system/user-management'),
    canEditUser: hasRoute('/system/user-management'),
    canDeleteUser: hasRoute('/system/user-management'),
    canResetPassword: hasRoute('/system/user-management'),

    canAddRole: hasRoute('/system/role-management'),
    canEditRole: hasRoute('/system/role-management'),
    canDeleteRole: hasRoute('/system/role-management'),

### 3. **在 System 组件中根据权限过滤导航菜单**

```typescript
// 示例思路
import { useAccess } from '@umijs/max';

const System = () => {
  const access = useAccess();
  const [selectedKey, setSelectedKey] = useState('user-profile');

  // 定义所有可能的菜单项
  const allMenuItems = [
    {
      key: 'user-profile',
      label: '个人信息',
      accessKey: 'canAccessUserProfile',
    },
    {
      key: 'user-management',
      label: '用户管理',
      accessKey: 'canAccessUserManagement',
    },
    {
      key: 'role-management',
      label: '角色管理',
      accessKey: 'canAccessRoleManagement',
    },
    {
      key: 'resource-management',
      label: '资源管理',
      accessKey: 'canAccessResourceManagement',
    },
  ];

  // 根据权限过滤菜单项
  const menuItems = allMenuItems.filter((item) => access[item.accessKey]);

  // 如果当前选中的菜单项被过滤掉了，选择第一个可用的菜单项
  useEffect(() => {
    if (
      !access[allMenuItems.find((item) => item.key === selectedKey)?.accessKey]
    ) {
      const firstAvailable = allMenuItems.find(
        (item) => access[item.accessKey],
      );
      if (firstAvailable) {
        setSelectedKey(firstAvailable.key);
      }
    }
  }, [access]);

  const renderContent = () => {
    switch (selectedKey) {
      case 'user-profile':
        return <UserProfile />;
      case 'user-management':
        return <UserManagement />;
      case 'role-management':
        return <RoleManagement />;
      case 'resource-management':
        return <ResourceManagement />;
      default:
        return null;
    }
  };

  // ... 其他代码
};
```

### 6. **更细粒度的权限控制（可选方案）**

如果需要更细粒度的权限控制（比如区分查看、新增、编辑、删除），有两种方案：

#### 方案一：后端返回更详细的路由权限

```typescript
// 后端返回格式示例
{
  routes: [
    '/system/user-management', // 可以访问用户管理页面
    '/system/user-management/add', // 可以新增用户
    '/system/user-management/edit', // 可以编辑用户
    '/system/user-management/delete', // 可以删除用户
    '/system/role-management',
    '/system/role-management/assign', // 可以分配权限
  ];
}
```

然后在 access.ts 中：

```typescript
// 示例思路
export default (initialState: API.UserInfo) => {
  const { routes = [] } = initialState || {};

  const hasRoute = (routePath: string) => routes.includes(routePath);

  return {
    // 页面级权限
    canAccessUserManagement: hasRoute('/system/user-management'),

    // 操作级权限
    canAddUser: hasRoute('/system/user-management/add'),
    canEditUser: hasRoute('/system/user-management/edit'),
    canDeleteUser: hasRoute('/system/user-management/delete'),

    canAccessRoleManagement: hasRoute('/system/role-management'),
    canAssignPermission: hasRoute('/system/role-management/assign'),
  };
};
```

#### 方案二：前端约定权限规则

```typescript
// 示例思路
export default (initialState: API.UserInfo) => {
  const { routes = [] } = initialState || {};

  const hasRoute = (routePath: string) => routes.includes(routePath);

  // 约定：如果有页面访问权限，就默认有该页面的所有操作权限
  // 或者约定：只有特定角色（如 admin）才有删除权限
  const isAdmin = hasRoute('/admin'); // 假设后端返回 '/admin' 表示管理员

  return {
    canAccessUserManagement: hasRoute('/system/user-management'),
    canAddUser: hasRoute('/system/user-management'),
    canEditUser: hasRoute('/system/user-management'),
    canDeleteUser: hasRoute('/system/user-management') && isAdmin, // 只有管理员可以删除
    canResetPassword: hasRoute('/system/user-management'),
  };
};
```
